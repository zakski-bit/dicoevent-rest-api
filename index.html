Production Infrastructure & Environment Configuration

Variabel lingkungan dan spesifikasi kontainer Docker untuk menjalankan backend secara utuh.

ENVIRONMENT VARIABLES (.env) 
# Django Configuration
DEBUG=True
SECRET_KEY=django-insecure-dicoevent-versii-2-production-secret-key
ALLOWED_HOSTS=*

# PostgreSQL Database
DATABASE_NAME=dicoevent_db
DATABASE_USER=postgres
DATABASE_PASSWORD=postgres
DATABASE_HOST=localhost
DATABASE_PORT=5432

# Redis Cache (Kriteria 2)
REDIS_HOST=localhost:6379

# Celery Asynchronous Tasks (Kriteria 3)
CELERY_BROKER_URL=redis://localhost:6379/0

# Email SMTP Credentials (Kriteria 3)
MAIL_HOST=smtp.gmail.com
MAIL_PORT=587
MAIL_USER=notification@dicoevent.com
MAIL_PASSWORD=secret_app_password

# MinIO Object Storage (Kriteria 1)
MINIO_ENDPOINT_URL=localhost:9000
MINIO_ACCESS_KEY=minioadmin
MINIO_SECRET_KEY=minioadmin
MINIO_SECURE=False
DOCKER COMPOSE STACK (docker-compose.yml) 
version: '3.8'

services:
  db:
    image: postgres:16-alpine
    container_name: dicoevent-postgres
    environment:
      POSTGRES_DB: dicoevent_db
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data

  redis:
    image: redis:alpine
    container_name: dicoevent-redis
    ports:
      - "6379:6379"

  minio:
    image: minio/minio:latest
    container_name: dicoevent-minio
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: minioadmin
      MINIO_ROOT_PASSWORD: minioadmin
    ports:
      - "9000:9000"
      - "9001:9001"
    volumes:
      - miniodata:/data
